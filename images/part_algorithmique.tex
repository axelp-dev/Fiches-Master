\documentclass[tikz,border=5pt]{standalone}
\usetikzlibrary{arrows,shapes,backgrounds}

\begin{document}

\tikzstyle{vertex}=[circle,fill=black!25,minimum size=20pt,inner sep=0pt]
\tikzstyle{edge} = [draw,thick,-]
\tikzstyle{weight} = [font=\small]

\begin{tikzpicture}[scale=1.8, auto,swap]
    % Draw vertices
    \foreach \pos/\name in {{(0,2)/a}, {(2,1)/b}, {(4,1)/c},
                            {(0,0)/d}, {(3,0)/e}, {(2,-1)/f}, {(4,-1)/g}}
        \node[vertex] (\name) at \pos {$\name$};

    % Connect vertices with edges and draw weights
    \foreach \source/ \dest /\weight in {b/a/7, c/b/8,d/a/5,d/b/9,
                                         e/b/7, e/c/5,e/d/15,
                                         f/d/6,f/e/8,
                                         g/e/9,g/f/11}
        \path[edge] (\source) -- node[weight] {$\weight$} (\dest);

    % Highlight selected edges (optional static version)
    \foreach \source / \dest in {d/a,d/f,a/b,b/e,e/c,e/g}
        \draw[draw,line width=2pt,red!50] (\source.center) -- (\dest.center);
    
    % Optionally mark the selected vertices
    \foreach \vertex in {d,a,f,b,e,c,g}
        \node[circle,fill=red!24,minimum size=20pt,inner sep=0pt] at (\vertex) {$\vertex$};
\end{tikzpicture}

\end{document}
