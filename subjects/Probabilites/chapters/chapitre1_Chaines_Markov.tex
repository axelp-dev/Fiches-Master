% Introduction 

Jusqu'à maintenant, nous avons étudié des suites de variables aléatoires dites iid (indépendantes et 
identiquement distribuées). Ces propriétés nous ont permis de développer beaucoup d'outils pour les étudier 
tels que la loi forte des grands nombres ou le théorème central limite. 

Or nous n'utilisons pas beaucoup ce type de suite de variables aléatoires en simulation. En effet, ces propriétés 
plutôt fortes ne permettent pas de modéliser des phénomènes dont l'évolution est plus complexe. Nous allons donc 
continuer à étudier des suites de variables aléatoires mais en rajoutant des relations de dépendance entre celles-ci. 
Cela nous permettra de modéliser des systèmes dynamiques plus complexes.

\section{Définitions et Généralités}

On se place ici dans un espace probabilisé $(E, \mathcal{F}, \myP)$  et $E$ un \emph{espace d'état}
fini ou dénombrable. On s'intéresse à une suite de variables aléatoires $(X_n)_{n \in \N}$ ou $X_k$ est 
l'état du système au temps $k$. 

\begin{definition}[Trajectoire]
    On appelle \emph{trajectoire du système} jusqu'au temps $n \in \N^*$ la donnée $(X_1, \dots, X_n)$. 
\end{definition}

\begin{definition}[Chaîne de Markov]
    On dit qu'une suite $(X_n)_{n \geqslant 0}$ est une \emph{chaîne de Markov} si elle 
    vérifie la \emph{propriété de Markov} : 
        \[ \forall n \in \N, \forall i_0, i_1, \dots, i_{n-1} \in E, \] 
        \[ \boxed{ \myP (X_{n+1} = j \; | \; X_n = i, X_{n-1} = i_{n-1}, \dots, X_0 = i_0) = \myP (X_{n+1} = j \; | \; X_n = i)} \]
\end{definition}

\begin{remark}
    Conditionnellement à l'état présent $\{X_n = i\}$, la probabilité d'arriver à l'état $\{X_{n+1} = j\}$ 
    à l'itération suivante ne dépend pas des évènements du passé $\{X_0 = i_0, \dots, X_{n-1} = n-1\}$. 
    On parle \emph{d'abscence de mémoire}. 
\end{remark}

\begin{definition}[Chaîne de Markov Homogène]
    Soit $(X_n)_{n \geqslant 0}$ une chaîne de Markov. On dit qu'elle est \emph{homogène} si : 
        \[ \forall n \in \N, \forall i,j \in E, \quad \myP (X_{n+1} = j \; | \; X_n = i) = \myP(X_1 = j \; | \; X_0 = i) \]
    On se placera toujours dans ce cadre.
\end{definition}

\begin{remark}
    Une chaîne de Markov homogène est une chaîne de Markov où la probabilité de passer d'un état à un autre ne dépend pas du temps. 
\end{remark}

\begin{example}[Marche Aléatoire]
    On modélise le mouvement d'une particule sur une grille par une chaîne de Markov où $X_n$ représente 
    la position de la particule au temps $n \in  \N$. On considère que la particule se déplace sur une 
    seule dimension. Elle peut donc avancer d'une case ou reculer d'une case à chaque itération. On notera 
    $ p \in [0,1]$ la probabilité qu'elle avance et $1-p$ la probabilité qu'elle recule indépendament du passé. 
    
    Soit $ (\varepsilon_i)_{i \geqslant 0}$ à valeurs dans $\{-1, 1\}$ telle que : 
        \[ \myP( \varepsilon_i = 1) = 1 - \myP( \varepsilon_i = -1) = p \in [0,1] \]
    la variable aléatoire qui représente le mouvement de la particule.  
    Déterminons la loi de $X_n$ :
        \[ 
            X_{n+1} = X_n + \varepsilon_{n+1} 
            = X_{n-1} + \varepsilon_n + + \varepsilon_{n+1} 
            = \dots 
            = \sum_{i=1}^{n} \varepsilon_i 
        \] 
    Montrons que c'est bien une chaîne de Markov : 
        \begin{align*}
            X_{n+1} &= X_n + \varepsilon_{n+1} \\ 
            &= \left( \sum_{i=1}^{n} \varepsilon_i \right) + \varepsilon_{n+1} \\ 
            &= \sum_{i=1}^{n+1} \varepsilon_i 
        \end{align*}
    où tous les $ \varepsilon_i, i \in \llbracket 1, n+1 \rrbracket $ sont indépendants. 
    On a donc : 
        \begin{align*}
            \myP(X_{n+1} = j \; | \; X_n = i, \dots, X_0 = i_0) &= \myP(X_n + \varepsilon_{n+1} = j \; | \; X_n = i, \dots, X_0 = i_0) \\ 
            &= \myP(i + \varepsilon_{n+1} = j \; | \; X_n = i, \dots, X_0 = i_0) \\ 
            &= \myP( \varepsilon_{n+1} = j - i) 
        \end{align*}

    On pourra alors s'intéresser au nombre de fois où une trajectoire passe par un point $n \in \N$. 
    La théorie des chaînes de Markov permet de répondre à ce genre de questions. 
\end{example}

Plus généralement, on peut construire récursivement des chaînes de Markov en définissant un $X_0$ fixé ou tiré par une 
loi $ \pi_0$ puis en posant : 
    \[ X_{n+1} = f(X_n, \varepsilon_{n+1}) \]
où $( \varepsilon_i)$ est une suite de variables aléatoires iid et $f$ une fonction mesurable. 

\begin{definition}[Matrice de Transition]
    Soit $(X_n)_{n \geqslant 0}$ une chaîne de Markov. On définit la \emph{matrice de transition} $P$ 
    de la chaîne comme la matrice, potentiellement infinie, telle que : 
        \[ P(i,j) = \myP(X_1 = j | X_0 = i) \] 
    Ainsi le coefficient de la i-ème ligne et de la j-ème colonne correspond à la probabilité de passer 
    de l'état $i$ à l'état $j$ en une étape.   
\end{definition}

\begin{definition}[Matrice Stochastique]
    La matrice de transition $P$ est dite \emph{stochastique} si 
        \[ \forall i \in  E, \quad \sum_{j \in E} P(i,j) = 1 \] 
    (i.e si la somme des poids des arrêtes de tout noeud du graphe associé est égale à 1)
\end{definition}

\begin{proposition}
    La loi d'une chaîne de Markov homogène $(X_n)_{n \geqslant 0}$ est complètement déterminée par la loi de $ \pi_0$ 
    de la condition initiale $X_0$ et par la matrice de transition $P$. On a : 
        \[ \forall i_0, i, \dots, i_n \in E, \quad \myP(X_0 = i_0, X_1 = i_1, \dots, X_n = i_n) = \pi_0(i_0)P(i_0, i_1)P(i_1, i_2) \dots P(i_{n-1}, i_n) \] 
\end{proposition}

\begin{proof}
    Démontrons ce résultat par récurrence sur $\N$. 
    \begin{itemize}
        \item \underline{Initialisation : } Pour $n = 0$, par définition, on a : 
            \[ \myP(X_0 = i_0) = \pi_0(i_0) \] 
        \item \underline{Hérédité : } Soient $i_0, \dots, i_n \in E$, supposons que : 
            \[ \myP(X_0 = i_0, X_1 = i_1, \dots, X_n = i_n) = \pi_0(i_0)P(i_0, i_1)P(i_1, i_2) \dots P(i_{n-1}, i_n) \] 
        Soit $i_{n+1} \in  E$ alors : 
            \begin{align*}
                & \myP(X_0 = i_0, X_1 = i_1, \dots, X_n = i_n, X_{n+1} = i_{n+1}) \\
                =& \myP(X_0 = i_0, X_1 = i_1, \dots, X_n = i_n) \times \myP(X_{n+1} = i_{n+1} | X_0 = i_0, \dots, X_n = i_n ) \\ 
                =& \myP(X_0 = i_0, X_1 = i_1, \dots, X_n = i_n) \times \myP(X_{n+1} = i_{n+1} | X_n = i_n) \\ 
                =& \pi_0(i_0)P(i_0, i_1)P(i_1, i_2) \dots P(i_{n-1}, i_n) P(i_n, i_{n+1})
            \end{align*}
    \end{itemize}
\end{proof}

On s'intéresse maintenant à la loi de $X_n$ au temps $n \in \N$. 

\begin{prop}[Relation de Chapman-Kolmogorov]
    Soit $(X_n)_{n \geqslant 0}$ une chaîne de markov homogène de matrice $P$ et de loi initiale $ \pi_0$. 
    Pour tout $i,j \in E$, on a : 
        \[ \myP(X_{n +m} = j | X_0 = i_0) = \sum_{k \in E} \myP(X_{n+m} = j | X_n = k) \times \myP(X_n = k | X_0 = i_0) \]
    En particulier : 
        \[ \myP(X_n = j | X_0 = i_0) = P^n (i_0, j) \]   
\end{prop}

\begin{proof}
    Prouvons les résultats précédents. 
    \begin{quote}
        \emph{
            Rappelons tout d'abord la formule des probabilités totales. 
            Soit $(B_k)_{k \in  \N}$ une partition de $E$, alors pour tout $A \in E$, on a :
                \[ \myP(A) = \sum_{k \in \N} \myP(A \cap B_k) \] 
            ou en conditionnel sur $C$ :
                \[ \myP(A|C) = \sum_{k \in \N} \myP(A \cap B_k | C) \]   
        }
    \end{quote}
    Or ici, $\{X_k | k \geqslant 0\}$ forme bien une partition de $E$ puisque à chaque instant $k$, 
    le processus est dans un unique état $X_k$. D'après la formule des probabilités totales, on a donc :
    \begin{align*}
        \myP(X_{n +m} = j | X_0 = i_0) &= \sum_{k \in E} \myP(X_{n+m} = j, X_n = k | X_0 = i_0) \\ 
        &= \sum_{k \in E} \myP(X_{n+m} = j | X_n = k, X_0 = i_0) \times \myP(X_n = k | X_0 = i_0) \\ 
        &=  \sum_{k \in E} \myP(X_{n+m} = j | X_n = k) \myP(X_n = k | X_0 = i_0)  
    \end{align*}
    On remarque de $\myP(X_{n+m} = j | X_n = k)$ est la probabilité de passer de l'état $k$ à $j$ en $m$ pas d'où :
        \[ \myP(X_{n+m} = j | X_n = k) = \myP(X_m = j | X_0 = k) \]   
    Déduisons maintetant la seconde formule de la première : 
        \begin{align*}
            \myP(X_2 = j | X_0 = i_0) &= \sum_{k \in E} \myP(X_2 = j | X_1 = k) \myP(X_1 = k | X_0 = i_0) \\ 
            &= \sum_{k \in E} P(i,k)P(k,i) \\ 
            &= P^2(i,j) \quad \text{etc } \dots    
        \end{align*}
\end{proof}

\begin{remark}[Notation]
    Si $X_0$ est distribué selon la loi $ \pi_0$, on notera : 
        \begin{align*}
            \myP_{ \pi_0} (X_n = j) &= \sum_{i_0 \in E} \myP(X_n = j | X_0 = i_0) \pi_0(i_0) \\ 
            &= \sum_{i_0 \in E} \myP(X_n = j | X_0 = i_0) \myP(X_0 = i_0) \\ 
            &= \sum_{i_0 \in  E} P(i_0, j) \pi_0(i_0) \\ 
            &= \pi_0 P^n(j) 
        \end{align*}
    où  $P^n(j) $ est la ième colonne de $P^n$ et $ \pi_0$ un vecteur ligne. 
\end{remark}

\begin{example}[Simple]
    Soit $(X_n)_{n \geqslant 0}$ une suite de variables aléatoires réelles iid de loi $ \pi$. 
        \[ \text{i.e} \quad \forall i \in E, \myP(X_n = i) = \pi(i) \] 
    On a donc : 
        \[ P(i,j) = \myP(X_1 = j | X_0 = i ) = \myP(X_1 = j) = \pi(j) \] 
    C'est un cas très simple et peu intéressant.
\end{example}

\begin{example}[Processus de Galton-Watson]
    Ici $(X_n)_{n \geqslant 1}$ compte le nombre d'individus d'une population dans la génération $n$. 
    On suppose que chaque individu vivant a des descendants indépendament des autres.

\end{example}

\section{Propriétés de Markov}

\subsection{Propriété de Markov Faible}

\begin{theorem}[Propriété de Markov Faible]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène de matrice de transition $P$ et de loi initiale 
    $ \pi_0$. Soit $ m \in \N$ fixé. Alors, conditionnellement à l’événement $\{X_m = i\}$, la chaîne décalée 
    $(X_{m+k})_{k \in \N}$ est une chaîne de Markov homogène de même matrice de transition $P$ et de loi initiale 
    $\delta_i$.

    \vspace{0.3cm}

    En particulier, le \emph{futur} $(X_{m+1}, X_{m+2}, \dots)$ est indépendant du \emph{passé} 
    $(X_0, X_1, \dots, X_{m-1})$ conditionnellement à $X_m$. Autrement dit, pour tous événements $A$ mesurables 
    par rapport à $\sigma(X_{m+1}, X_{m+2}, \dots)$ et $B$ mesurables par rapport à 
    $\sigma(X_0, X_1, \dots, X_{m-1})$, on a :
    
    \[
        \mathbb{P}(A \cap B \mid X_m = i)
        = \mathbb{P}(A \mid X_m = i)\,\mathbb{P}(B \mid X_m = i).
    \]
\end{theorem}

\begin{proof}
    Démontrons ce théorème en deux parties : 

    \begin{enumerate}
        \item \underline{Égalité des lois :} 

        Montrons que, pour tout $n \in \N$ et toute suite d’états $(y_1, \dots, y_n)$,
        \[ \myP(X_{m+1} = y_1, \dots, X_{m+n} = y_n \mid X_m = i, X_{m-1} = i_{m-1}, \dots, X_0 = i_0) \]
        \[ = \myP(X_1 = y_1, \dots, X_n = y_n \mid X_0 = i) \] 
        On a :
        \begin{align*}
            &\myP(X_{m+1} = y_1, \dots, X_{m+n} = y_n \mid X_m = i, X_{m-1} = i_{m-1}, \dots, X_0 = i_0) \\[0.8em]
            &= \frac{
                \myP(X_{m+1} = y_1, \dots, X_{m+n} = y_n, X_m = i, X_{m-1} = i_{m-1}, \dots, X_0 = i_0)
            }{
                \myP(X_m = i, X_{m-1} = i_{m-1}, \dots, X_0 = i_0)
            } \\[0.8em]
            &= \frac{
                \pi_0(i_0) P(i_0, i_1) \cdots P(i_{m-1}, i)
                P(i, y_1) \cdots P(y_{n-1}, y_n)
            }{
                \pi_0(i_0) P(i_0, i_1) \cdots P(i_{m-1}, i)
            } \\[0.8em]
            &= P(i, y_1) P(y_1, y_2) \cdots P(y_{n-1}, y_n) \\[0.8em]
            &= \myP(X_1 = y_1, \dots, X_n = y_n \mid X_0 = i).
        \end{align*}
        Ainsi, la chaîne décalée $(X_{m+k})_{k \in \N}$, conditionnellement à $X_m = i$, 
        suit bien la loi d’une chaîne de Markov homogène de matrice $P$ et de loi initiale $\delta_i$.

        \vspace{1em}

        \item \underline{Indépendance conditionnelle :}

        Soient $A$ un événement mesurable par rapport au futur $\sigma(X_{m+1}, X_{m+2}, \dots)$ 
        et $B$ un événement mesurable par rapport au passé $\sigma(X_0, X_1, \dots, X_{m-1})$.

        Montrons que :
        \[
            \myP(A \cap B \mid X_m = i)
            = \myP(A \mid X_m = i)\,\myP(B \mid X_m = i).
        \]

        Par définition de la probabilité conditionnelle :
        \[
            \myP(A \cap B \mid X_m = i)
            = \frac{\myP(A \cap B, X_m = i)}{\myP(X_m = i)}.
        \]
        D’après la propriété de Markov, conditionnellement à $X_m = i$, le futur $(X_{m+1}, X_{m+2}, \dots)$ 
        ne dépend pas du passé $(X_0, \dots, X_{m-1})$, donc :
        \[
            \myP(A \cap B, X_m = i)
            = \myP(A, X_m = i)\,\myP(B, X_m = i) / \myP(X_m = i).
        \]
        Ainsi :
        \begin{align*}
            \myP(A \cap B \mid X_m = i)
            &= \frac{\myP(A, X_m = i)\,\myP(B, X_m = i)}{\myP(X_m = i)^2} \\[0.5em]
            &= \myP(A \mid X_m = i)\,\myP(B \mid X_m = i).
        \end{align*}
        Ce qui montre bien l’indépendance conditionnelle du passé et du futur, sachant $X_m$.
    \end{enumerate}
\end{proof}

\begin{remark}
    Cette propriété de Markov est aussi vraie de façon plus générale quand on conditionne par l'état de la chaîne en un 
    temps aléatoire appelé temps d'arrêt. Ce sera exactement le propos de la propriété de Marjov forte que nous verrons 
    plus tard. 
\end{remark}

\subsection{Temps d'arrêt et propriété de Markov Forte}

\begin{definition}[Tribu $ \mathcal{F}_n$]
    On définit $ \mathcal{F}_n$ la tribu engendrée par les variables aléatoires $(X_1, \dots, X_n)$ la tribu : 
        \begin{align*}
            \mathcal{F}_n &= \sigma \left(X_0, \dots, X_n\right) \\ 
            &= \sigma \left( \{X_0 = x_0, \dots, X_n = x_n\}, x_0, \dots, x_n \in E\right) 
        \end{align*} 
    Ainsi, la tribu engendrée par l'ensemble $ \{X_0 = x_0, \dots, X_n = x_n\}$ est la plus petite tribu 
    contenant cette ensemble. 
\end{definition}

\begin{definition}[Temps d'arrêt]
    Soit $(X_n)_{n \geqslant 0}$ une chaîne de Markov à valeurs dans un espace d'états $E$, définie sur un espace probabilisé $(\Omega, \mathcal{F}, \mathbb{P})$, avec 
        \[ \forall n \in \N, \quad X_n : \Omega \longrightarrow E \]
    On note $\mathcal{F}_n = \sigma(X_0, X_1, \dots, X_n)$ la tribu engendrée par les $n$ premières variables de la chaîne.
    Une application
        \[ \boxed{\tau : \Omega \longrightarrow \N \cup \{\infty\}} \]
    est appelée \emph{temps d'arrêt} de la chaîne de Markov (ou pour la filtration $(\mathcal{F}_n)_{n \geqslant 0}$) si elle vérifie :
        \[ \boxed{ \forall n \in \N, \quad \{ \tau \leqslant n \} \in \mathcal{F}_n} \]
    Donc $ \tau$ est mesurable pour la tribu $ \mathcal{F}_n$.  
    Autrement dit, le fait que l'évènement $\{\tau \leqslant n\}$ ne dépend que de l’évolution de la chaîne jusqu’au temps $n$.
\end{definition}

\begin{example}
    Soit $(X_n)$ une chaîne de Markov sur $\Z$. On définit le temps d'attente du dépassement de $ j \in \Z$ comme 
    la variable aléatoire : 
        \[ \tau_i = \min \{i \geqslant 0 | X_i > j\} \] 
    $ \tau_j$ est bien un temps d'arrêt pour la CMH car : 
        $$ \{ \tau_j \leqslant n\} = \{ \exists k \in \N, X_k \geqslant j\} $$
    donc $ \tau_j$ ne dépend que de $X_0, \dots, X_n$. 
    De façon générale, soit $A \subset E$, le temps d'attente de $A$ défini par : 
        \[ \tau_A = \min \{i \geqslant 0, X_i \in A\} \]
    est un temps d'arrêt. Si cet ensemble est vide (i.e la chaîne de Markov n'atteint jamais $A$), alors $ \tau_A = + \infty$. 
    Par contre, $ S_j = \max \{i \geqslant 0, X_i > j\}$ n'est pas un temps d'arrêt car 
    $ \{S_j \leqslant n\} = {X_{n+1} < j, X_{n+2} < j, \dots}$ dépend de ce qui s'est passé "après". 
\end{example}

\begin{theorem}[Propriété de Markov forte]
    Soit $(X_n)_{n \geqslant 0}$ une chaîne de Markov homogène et $ \tau$ un temps d'arrêt. 
    Alors conditionnellement à $\{\tau < \infty\}$ et $\{X_\tau = x\}$ la chaîne décalée $(X_{\tau+n})_{n \geqslant 0}$ 
    est encore une chaîne de Markov de condition initiale $x$. 

    De plus, conditionnellement à $\{\tau < \infty\} \cap \{X_\tau = x\}$ on a : 
        \[ (X_{\tau+n})_{n \geqslant 0} \text{ est indépendante de } (X_0, \dots, X_{\tau-1}) \]  
\end{theorem}

\begin{remark}
    Ici, la chaîne de Markov est décalée conditionnellement à une variable aléatoire, c'est beaucoup plus fort 
    que le propriété précédente. 
\end{remark}

\begin{example}[Application - Ruine du Joueur]
    Soit l'espace d'état $E = \{ 0, \dots, m \}$ et $(X_n)_{n \in \N}$ une chaîne de Markov sur $E$ 
    telle que $X_0 = k \in \{ 1, \dots, m \}$. On s'intéresse à deux temps d'arrêts : 
    \begin{itemize}
        \item $ \tau_0 = \inf \{ n \geqslant 0 \; | \; X_n = 0 \} $ 
        \item $  \tau_m = \inf \{ n \geqslant 0 \; | \; X_n = m \}$ 
    \end{itemize} 
    On souhaite estimer le temps de jeu moyen d'un joueur. Pour cela, on considère le temps d'arrêt : 
        $$ \tau = \min ( \tau_0, \tau_m) $$ 
    On veut savoir quand $t < t_0$ (i.e quand le joueur gagne avant de perdre). 
    Soit $ m> 1$, alors $ \tau_0$ est distinct de $ \tau_m$ si l'un des deux est fini. En conséquence 
    si $ \tau < \infty$, on a : 
    \begin{itemize}
        \item soit $ \tau = \tau_0 < \tau_m$ 
        \item soit $ \tau = \tau_m < \tau_0$. 
    \end{itemize}
    On cherche la "probabilité de ruine" : 
        $$ f(k) = \myP( \tau < \infty, \tau = \tau_0 \; | \; X_0 = k) $$ 
    Pour cela, nous devons donc déterminer toutes les valeurs de $f$. On utilise donc les propriétés de Markov pour en déduire 
    des relations de récurrence entre $f(k), f(k+1)$ et $f(k-1)$. On remarque ainsi que : 
        $$ \text{si } X_0 = 0 \text{ alors } \tau = \tau_0 = 0 \text{ donc } f(0) = 1 $$ 
        $$ \text{si } X_0 = m \text{ alors } \tau = \tau_m = 0 \text{ donc } f(m) = 0 $$
    Si $X_0 \not \in \{ 0, m \} $, on doit au moins faire un pas pour atteindre $ \tau_0$ ou $ \tau_m$. On en déduis donc que : 
        $$ \tau_0((X_n)_{n \in \N}) = 1 + \tau_0((X_{n+1})_{n \in \N}) $$ 
        $$ \tau_m((X_n)_{n \in \N}) = 1 + \tau_m((X_{n+1})_{n \in \N}) $$ 
    On a donc, via l'analyse à un pas : 
        \begin{align*}
            f(k) &= \myP( \tau((X_n)_{n \in \N}) = \tau_0((X_n)_{n \in \N}) < \infty \; | \; X_0 = k) \\
            &= \myP( \tau((X_{n+1})_{n \in \N}) = \tau_0((X_{n+1})_{n \in \N}) < \infty \; | \; X_0 = k)
        \end{align*}
    En appliquant la formule des probabilités totales à $X_1$, on a donc : 
        \begin{align*}
            f(k) &= \myP(X_1 = k+1 \; | \; X_0 = k) \times \myP( \tau((X_{n+1})_{n \in \N}) = \tau_0((X_{n+1})_{n \in \N}) < \infty \; | \; X_1 = k+1, X_0 = k) \\ 
            & \quad + \myP(X_1 = k-1 \; | \; X_0 = k) \times \myP( \tau((X_{n+1})_{n \in \N}) = \tau_0((X_{n+1})_{n \in \N}) < \infty \; | \; X_1 = k-1, X_0 = k) 
        \end{align*}
    Or, conditionnellement à $ \{ X_1 = k+1 \}$, la chaîne $ (X_{n+1})_{n \in \N}$ est une chaîne de Markov 
    partant de $k+1$. On a donc : 
        \begin{align*}
            f(k) &= p \myP( \tau_0((X_n)_{n \in \N}) = \tau((X_n)_{n \in \N}) \; | \; X_0 = k+1) \\ 
            & \quad + (1-p) \myP( \tau_0((X_n)_{n \in \N}) = \tau((X_n)_{n \in \N}) \; | \; X_0 = k-1) \\ 
            &= pf(k+1) + (1-p)f(k-1)
        \end{align*}
    Résolvons cette relation de récurrence pour $p = 1/2$. On a donc $ \forall k \in \llbracket 1, m \rrbracket, 
    k \not \in \{ 0,m \} $ : 
        $$ f(k) = \frac{f(k+1) + f(k-1)}{2} $$ 
        $$ \iff f(k+1) - f(k) = f(k) - f(k-1) = \Delta$$ 
    On peut donc poser : 
        \begin{align*}
            & f(m) - f(0) = \sum_{k=0}^{m-1} f(k+1) - f(k) = (m-1) \Delta \\ 
            & \iff \Delta = - \frac{1}{m}
        \end{align*}
    D'où : 
        $$ f(k) = 1 + k \Delta = 1 - \frac{k}{m} $$
\end{example}

\section{Mesures Invariantes}

Dans cette section, nous supposons tout d'abord que l'espace d'états $E$ est fini. 
Au début du cours on a vu que, pour une chaîne de Markov $(X_n)_{n \in \N}$ de matrice $P$
et de loi initiale $ \pi_0$, la relation de Chapman-Kolmogorov nous donne : 
    $$ \myP(X_n = j \; | \; X_0 = i_0) = P^n(i_0, j) $$ 
Nous allons ici chercher des mesures de probabilités telles que : 
    $$ \pi P = \pi $$
Nous appellerons de telles mesures des \emph{mesures de probabilités invariantes}. 

\subsection{Définition et propriétés}

\begin{definition}[Mesures Invariantes]
    Soit une mesure positive $ \pi$ sur $E$. On dit que $ \pi$ est \emph{invariante} pour une matrice de transition 
    $P$ si : 
        $$ \boxed{\forall x \in E, \quad \pi(x) = \pi P(x) = \sum_{y \in E} \pi(y)P(y,x)} $$
    On dit alors que $ \pi$ est un \emph{vecteur propre à gauche} de $P$ pour la valeur propre 1. 
    Par convention, les mesures sont écrites comme des vecteurs ligne. 
\end{definition}

\begin{remark}
    Attention, une mesure invariante n'est pas forcément une mesure de probabilité. Pour que ce 
    soit le cas, il faut que sa masse totale soit égale à $1$. 
        $$ \text{i.e} \quad \sum_{x \in E} \pi(x) = 1 $$
    Or ici, puisque $E$ est fini, cette somme l'est aussi donc toute mesure invariante peut être normalisée
    en mesure de probabilités. Nous verrons plus tard que ce n'est pas toujours le cas. 
\end{remark}

\begin{lemma}
    Soit $ \pi$ une mesure invariante pour une matrice de transition $P$. Alors : 
        $$ \forall n \in \N^*, \quad \pi = \pi P^n $$ 
    De plus si $X_0$ est distribuée selon la loi $ \pi$ alors la loi de $X_n$ l'est également 
    pour tout $ n > 0$. En conséquence : 
        $$ \forall n,m \in N, \quad (X_0, X_1, \dots, X_m) \overset{ \text{loi} }{=} (X_n, X_{n+1}, \dots, X_{n+m}) $$
\end{lemma}

\begin{prop}[Mesure Invariante et Loi]
    Soit $ \pi$ une mesure de probabilité invariante. Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène de 
    matrice $P$ et de loi initiale $ \pi_0$ alors, pour tout $n \in \N, X_n \sim \pi$. 
\end{prop}

Ce lemme justifie toute l'utilisation que nous ferons des mesures invariantes. En effet, une mesure invariante 
permet de déterminer le comportement asymptotique de la chaîne de Markov sans avoir à la simuler en entier un 
grand nombre de fois. 

\subsection{Existence des mesures invariantes}

\begin{theorem}[Mesure invariance, existence]
    Toute chaîne de Markov homogène sur un espace d'état fini admet une mesure de probabilité invariante. 
\end{theorem}

\begin{proof}
    Soit $ \psi$ une mesure de probabilité quelconque sur $E$ où $ |E| = d$. Posons : 
        $$ \forall n \in \N, \quad \psi = \frac{1}{n} \sum_{k=0}^{n} \psi P^k $$ 
    On sait que pour tout $k \in \N, \psi P^k$ est une mesure de probabilité sur $E$. Donc $ \psi_n$ l'est aussi 
    une par construction. 
    
    De plus, pout tout $x \in E, ( \psi_n(x)_{n \in \N})$ est une suite à valeurs dans $[0,1]^d$. Par compacité,
    il existe une sous-suite convergente $( \psi_{ \varphi(n)}(x))_{n \in \N}$. Comme $E$ est fini, il existe une 
    extraction telle que : 
        $$ \forall x \in E, \quad \psi_{ \varphi(n)}(x) \underset{n \to \infty}{\longrightarrow} \pi(x) $$
    Montrons maintenant que $ \pi$ est une mesure de probabilités invariante pour $P$. 
    On a : 
        $$ \forall n \in \N, \quad \sum_{x \in E} \psi_n(x) = 1 \quad \text{et} \quad \forall x \in E, \quad \psi_n(x) \geqslant 0 $$
    Donc lorsque $ n \to \infty$, on a :
        $$ \sum_{x \in E} \pi(x) = 1 \quad \text{ et } \quad \forall x \in E, \pi(x) \geqslant 0 $$
    Donc $ \pi$ est une mesure de probabilités sur $E$. De plus, pour tout $n \in \N$, on a : 
        \begin{align*}
            \psi_n P &= \frac{1}{n} \sum_{k=1}^{n} \psi P^k P =  \frac{1}{n} \sum_{k=1}^{n} \psi P^{k+1} = \frac{1}{n} \sum_{k=2}^{n+1} \psi P^k \\ 
            &= \frac{1}{n} \frac{n+1}{n+1} \left( \sum_{k=1}^{n+1} \psi P^k - \psi P \right) \\ 
            &= \frac{n+1}{n} \psi_{n+1} - \frac{1}{n} \psi P     
        \end{align*} 
    Donc lorsque $ n \to \infty$, on obtient : 
        $$ \pi P = \pi $$
\end{proof}

\begin{remark}
    Il existe pleins de méthodes numériques pour approcher une mesure invariante d'une chaîne de Markov homogène. 
    Ce n'est pas un problème facile car il faut en général résoudre un système de $d$ équations à $d$ inconnues. 
    Nous nous intéresserons donc aux \emph{mesures réversibles}, plus faciles à calculer et qui nous donnent 
    une mesure invariante quand elle existe. 
\end{remark}

\begin{example}
    Soit le graphe suivant et $ p,q \in ]0,1[$. 
    \begin{figure}[h!]
        \centering
        \makebox[\textwidth][c]{ % Centre le tikz dans la largeur du texte
            \begin{tikzpicture}[
                baseline=(current bounding box.center),
                shorten >=1pt,
                node distance=3cm,
                on grid,auto
            ]
                \node[state] (1) {$1$}; 
                \node[state, right of=1] (2) {$2$}; 

                \path[->]
                    (1) edge[above, bend left] node{$p$} (2)
                    (2) edge[loop right] node{$1-q$} (2)
                    (2) edge[above, bend left] node{$q$} (1)
                    (1) edge[loop left] node{$1-p$} (1); 
            \end{tikzpicture}
        }
        \caption{Mesure Invariante}
    \end{figure}
    Soit $ \pi = ( \pi_1, \pi_2)$ une mesure invariante. On a alors $ \pi P = \pi$. 
        \begin{align*}
            \iff & ( \pi_1, \pi_2) 
                \begin{pmatrix}
                    1-p & p \\ 
                    q & 1-q 
                \end{pmatrix} 
                = ( \pi_1, \pi_2) \\ 
            \iff & 
                \begin{cases}
                    \pi_1 (1-p) + q \pi_2 = \pi_1 \\ 
                    \pi_1 p + \pi_2(1-q) = \pi_2 
                \end{cases}\\ 
            \iff & 
                \pi_1 = \frac{q}{p} \pi_2 
        \end{align*}
        Or $ \pi$ est une mesure de probabilité ssi : 
            $$ pi_1 + \pi_2 = 1 \iff \pi_1 = \frac{q}{p+q} \text{ et } \pi_2 = \frac{p}{p+q} $$  
\end{example}

\subsection{Unicité des mesures de probabilité invariantes et irréductibilité}

\begin{definition}[Communication]
    Soient $x,y \in E$. On dit que \emph{$x$ communique avec $y$} s'il existe un chemin de probabilité positive 
    de $x$ à $y$ dans $E$. On le note alors $ x \leadsto x$. Plus formellement :
        \begin{align*}
            x \leadsto y & \iff \exists x, x_1, \dots, x_{n-1}, y \in E, P(x, x_1), P(x_1, x_2), \dots, P(x_{n-1}, y) > 0 \\ 
            & \iff \forall n \geqslant 1, \quad P^n(x,y) > 0 
        \end{align*}
\end{definition}

\begin{prop}[Transitivité]
    La relation communique $ \leadsto$ est transitive. 
\end{prop}

\begin{definition}[Classe fermée]
    Soit $E_0 \subset E$ un sous-ensemble de $E$ que l'on appellera \emph{classe de $E$}. 
    On dit que la classe $E_0$ est \emph{fermée} si : 
        $$ \forall x \in E_0, \quad x \leadsto y \Longrightarrow y \in E_0 $$ 
\end{definition}

\begin{definition}[Classe irréductible]
    Une classe $E_0$ de $E$ est dite \emph{irréductible} si elle est fermée et si : 
        $$ \forall x,y \in E_0, \quad x \leadsto y \text{ et } y \leadsto x $$ 
    Par extension, on dira qu'une chaîne de Markov homogène est irréductible si son espace d'état l'est aussi. 
\end{definition}

\begin{theorem}[Unicité des mesures invariantes]
    Toute chaîne de Markov homogène irréductible sur un espace d'état fini admet une \emph{unique mesure de probabilité 
    invariante} $ \pi$ et : 
        $$ \forall x \in E, \quad \pi(x) > 0 $$
\end{theorem}

\begin{proof}
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène d'espace d'état $E$. 
    Supposons que $E$ est irréductible et soit $ \pi$ une mesure invariante de $(X_n)_{n \in \N}$. 
    \begin{enumerate}
        \item \underline{Positivité :} Montrons que $ \pi$ est positive. On sait que $ \pi$ est une 
        mesure de probabilités donc : 
            $$ \sum_{x \in E} \pi(x) = 1 $$ 
        donc il existe nécessairement $x_0 \in E$ tel que $ \pi(x_0) > 0$. Comme $E$ st irréductible, 
        alors pour tout $x \in E, x_0 \leadsto y$. Donc il existe $ n \in \N$ tel que : 
            $$ P^n(x_0, y) > 0$$ 
        De plus, on sait que $ \pi$ vérifie : 
            \begin{align*}
                \pi(x) &= \sum_{x \in E} \pi(x) P(x,y) = \pi P \\ 
                \Longrightarrow \pi(y) &= \sum_{x \in E} \pi(x) P^n(x,y) \\ 
                &= \underbrace{\pi(x_0) P^n(x_0, y)}_{>0}
                + \underbrace{\sum_{x \in E, \, x \not= x_0} \pi(x) P^n(x,y)}_{\geqslant 0}
            \end{align*}
        Donc $ \forall y \in E, \pi(y) > 0$. 
        \item \underline{Unicité :} voir TD
    \end{enumerate}
\end{proof}

\subsection{Construction de mesures de probabilité invariantes}

Nous avons vu que les mesures de probabilité invariantes jouent un rôle fondamental dans l’étude des chaînes de Markov. 
Elles permettent notamment de décrire le comportement stationnaire du système. 
Nous allons à présent présenter quelques outils permettant de construire de telles mesures.

Soit $E$ un espace d’états fini, et $(X_n)_{n \in \N}$ une chaîne de Markov homogène et irréductible sur $E$.  
L’idée est de chercher à faire converger la loi de $X_n$ lorsque $n$ tend vers l’infini, de façon à obtenir une mesure de probabilité invariante pour la matrice de transition associée.

\begin{definition}[Temps de retour]
    Soit $x \in E$, on définit le temps de retour en $x$ comme : 
        $$ \boxed{\tau_x^+ := \inf \{ n \geqslant 1 \; | \; X_n = x \} \in \overline{\N}} $$ 
    remarquons que c'est un \emph{temps d'arrêt}.  
\end{definition}

\begin{remark}
    Nous pouvons facilement montrer que tout temps de retour est un temps d'arrêt. 
    En effet, pour rappel nous avons définit les temps d'arrêt comme : 
        \begin{quote}
            \emph{Une application
                \[ {\tau : \Omega \longrightarrow \N \cup \{\infty\}} \]
            est appelée \emph{temps d'arrêt} de la chaîne de Markov (ou pour la filtration $(\mathcal{F}_n)_{n \geqslant 0}$) si elle vérifie :
                \[ { \forall n \in \N, \quad \{ \tau \leqslant n \} \in \mathcal{F}_n} \]}
        \end{quote}
    Soit $n \in \N$ on a alors : 
        \begin{align*}
            \{ \tau_x^+ \leqslant n\} &= \{ \exists k \in \llbracket 1, n \rrbracket \; | \; X_k = x\} \\ 
            &= \bigcup_{k = 1}^{n} \{X_k = x\}
        \end{align*}
    or, pour tout $ k \in \llbracket 1, n \rrbracket $, l'évènement $ \{X_k = x\} \in \mathcal{F}_k \subseteq \mathcal{F}_n$. 
    Donc par union dénombrable, on a : 
        $$ \{ \tau_x^+ \leqslant n\} \in \mathcal{F}_n $$ 
    Donc tout temps d'attente est un temps d'arrêt. De plus, on peut remarque que la différence avec un temps de retour 
    est que si $ \{ X_0 = x \}$ alors : 
            $$ \tau_x = 0 \quad \text{et} \quad \tau_x^+ \geqslant 1 $$  
\end{remark}

\begin{lemma}
    Pour tout $x,y \in E$, on a :   
        $$ \mathbb{E}( \tau_y^+ \; | \; X_0 = x) < \infty $$ 
        % $$ \iff \sum_{k=1}^{ \infty} k \myP( \tau_y^+ = k \; | \; X_0 = x) + \infty   
\end{lemma}

\begin{theorem}[Mesure Invariante]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène sur un espace d'états $E$ fini. 
    Alors son unique mesure de probabilités invariante est : 
        \[ \boxed{\pi(x) = \frac{1}{ \mathbb{E}( \tau_x^+ | X_0 = x) } \quad \forall x \in E} \]  
\end{theorem}

\begin{remark}
    Cette mesure charge plus les points $x \in E$ tels que $ \mathbb{E}( \tau_x^+ | X_0 = x)$ est petit. 
    C'est à dire les points que la chaîne de Markov visite souvent.  
\end{remark}

\begin{proof}
    Soit $x \in E$ une condition initiale. Soit $ \overline{ \pi} : E \longrightarrow \N$ une application qui compte 
    le nombre de fois que la chaîne visite un point $ y \in E$. 
        \begin{align*}
            \forall y \in E, \quad \overline{ \pi}(y) &:= \mathbb{E} \left( \sum_{n=0}^{ \tau_x^+ - 1} \mathbbm{1}_{X_n = y} \; | \; X_0 = x \right) \\ 
            & = \mathbb{E} \left( \sum_{n \geqslant 0} \mathbbm{1}_{X_n = y} \times \mathbbm{1}_{ \tau_x^+ > n} \; | \; X_0 = x \right) \\ 
            &= \sum_{n \geqslant 0} \myP(X_n = y, \tau_x^+ > n \; | \; X_0 = x)  
        \end{align*}
    On veut montrer que $ \overline{ \pi}$ est invariante.  On veut donc : 
        \[ \forall y \in E, \quad \overline{ \pi}(y) = \sum_{z \in E} \overline{ \pi}(z) P(z, y) \]
    (On va également montrer que $ \overline{ \pi}$ de dépend pas de la condition initiale $x$.)
    Pour tout $z \in E$, calculons donc : 
        \[ \sum_{z \in E} \overline{ \pi}(z) P(y,z) = \sum_{ z \in E} \sum_{n \geqslant 0} \myP(X_n = z, \tau_x^+ > n \; | \; X_0 = x) \times P(z,y)  \] 
    Or on a : 
        \begin{align*}
            & \myP(X_n = z, \tau_x^+ > n, X_{n+1} = y \; | \; X_0 = x) \\
            &= \myP(X_{n+1} = y \; | \; X_n = z, \tau_x^+ > n, X_0 = x) \times \myP(X_n = z, \tau_x^+ > n, X_0 = x) \\
            &= \myP(X_{n+1} = y \; | \; X_n = z) \times \myP(X_n = z, \tau_x^+ > n, X_0 = x) \quad \quad \text{(Propriété de Markov)} \\ 
            &= P(z,y) \times \myP(X_n = z, \tau_x^+ > n, X_0 = x) 
        \end{align*}
    On en déduit alors que : 
        \begin{align*}
            \sum_{z \in E} \overline{ \pi}(z) P(y,z) &= \sum_{z \in E} \sum_{n \geqslant 0} \myP(X_n = z, \tau_x^+ > n, X_{n+1} = y \; | \; X_0 = x) \\ 
            &= \sum_{ n \geqslant 0} \myP(\tau_x^+ > n, X_{n+1} = y \; | \; X_0 = x) \quad \quad \text{(Probabilités Totales)} \\ 
            &= \sum_{ n \geqslant 0} \myP(\tau_x^+ \geqslant n +1, X_{n+1} = y \; | \; X_0 = x) \\ 
            &= \sum_{n \geqslant 1} \myP(\tau_x^+ \geqslant n, X_{n} = y \; | \; X_0 = x) \quad \quad \text{(Changement d'indices)} \\ 
            &= \sum_{n \geqslant 1} \myP(\tau_x^+ > n, X_{n} = y \; | \; X_0 = x) + \sum_{n \geqslant 1} \myP(\tau_x^+ = n, X_{n} = y \; | \; X_0 = x) \\ 
            &= \overline{ \pi}(y) - \myP(X_0 = y, \tau_x^+ > 0 \; | \; X_0 = x) + \sum_{n \geqslant 1} \myP(\tau_x^+ = n, X_{n} = y \; | \; X_0 = x)
        \end{align*}
    Or : 
        \begin{align*}
            & \sum_{n \geqslant 1} \myP(\tau_x^+ = n, X_{n} = y \; | \; X_0 = x) \\ 
            &= \sum_{n \geqslant 1} \myP(\tau_x^+ = n, X_{ \tau_x^+} = y \; | \; X_0 = x) \\ 
            &= \sum_{n \geqslant 1} \myP( \tau_x^+ < \infty, X_{ \tau_x^+} = y \; | \; X_0 = x) \\ 
            &= \myP(X_{ \tau_x^+} = y\; | \; X_0 = x)
        \end{align*}
    D'où : 
        \begin{align*}
            \sum_{z \in E} \overline{ \pi}(z) P(z,y) &= \overline{ \pi}(y) - \underbrace{\myP(X_0 = y, \tau_x^+ > 0 \mid X_0 = x)}_{1 \text{ si } y = x \text{ et } 0 \text{ sinon}} + \underbrace{ \myP(X_{ \tau_x^+} = y \mid X_0 = x)}_{-1 \text{ si } y = x \text{ et } 0 \text{ sinon}}
            = \overline{ \pi}(y)  
        \end{align*}
    
    Donc $ \overline{ \pi}$ est bien une mesure invariante de masse totale : 
        \[ \sum_{y \in E} \overline{ \pi}(y) = \mathbb{E}( \tau_x^+ \; | \; X_0 = x) \] 
    Or, d'après le lemme précédent, $  \mathbb{E}( \tau_x^+ \; | \; X_0 = x) < \infty$. 
    Donc l'application : 
        \[ \forall y \in E, \quad \pi(y) = \frac{ \overline{ \pi}(y)}{ \mathbb{E}(\tau_x^+ \; | \; X_0 = x)}\] 
    est bien une mesure de probabilités invariante. D'après le théorème d'unicité, c'est donc la seule. De plus, on a 
    $ \overline{ \pi}(x) = 1$ donc : 
        \[ \forall y \in E, \quad \pi(y) = \frac{ \overline{ \pi}(y)}{ \mathbb{E}( \tau_x^+ \; | \; X_0 = x)} = \frac{ 1}{ \mathbb{E}( \tau_x^+ \; | \; X_0 = x)} \] 
\end{proof}

\subsection{Réversibilité}

Trouver la mesure de probabilité invariante en résolvant le système est compliqué en général. La \emph{réversibilité}
est une propriété suffisante pour qu'une mesure soit invariante. 

\begin{definition}[Mesure réversible]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène de matrice $P$. Une mesure $ \mu$ est dite 
    réversible pour la matrice $P$ si : 
        \[ \boxed{\forall x,y \in E, \quad \mu(x) P(x,y) = \mu(y) P(y,x)} \] 
\end{definition}

\begin{remark}
    La réversibilité peut être vue comme le fait d'être capable de parcourir les trajectoires d'une chaîne de Markov 
    dans le sens inverse. 
        $$ \text{i.e} \quad \myP(X_0 = x_0, X_1 = x_1, \dots, X_n = x_n) = \myP(X_0 = x_n, X_1 = x_{n-1}, \dots, X_n = x_0) $$ 
    Ce résultat provient de la relation de Chapman-Kolmogorov. En effet, si $X_0$ est choisie selon $ \mu$ réversible, 
    alors : 
        \begin{align*}
            \myP(X_0 = x_0, X_1 = x_1, \dots, X_n = x_n) &= \mu(x_0) P(x_1, x_1) \dots P(x_{n-1}, x_n) \\ 
            &= P(x_1, x_0) \mu(x_1) P(x_1, x_2) \dots P(x_{n-1}, x_n) \\
            &= P(x_1, x_0) P(x_2, x_1) \dots P(x_n, x_{n-1}) \mu(x_n) \\ 
            &= \mu(x_n) P(x_n, x_{n-1}) \dots P(x_2, x_1) P(x_1, x_0) \\ 
            &= \myP(X_0 = x_n, X_1 = x_{n-1}, \dots, X_n = x_0)
        \end{align*}
\end{remark}

\begin{proposition}
    Une mesure réversible pour une matrice de transition est invariante. 
\end{proposition}

L'étude des mesures réversibles, plus simples que les mesures invariantes, sera donc privilégiée pour déterminer 
des mesures invariantes.

\section{Classification des états}

Ici, on suppose maintenant que $E$ est un espace d'états fini ou dénombrable et $(X_n)_{n \in \N}$ une chaîne 
de Markov homogène sur $E$ de matrice de transition $P$. 

\subsection{Nombre de visites en un point}

\begin{definition}[Nombre de visites]
    Soit $x \in E$, on note $V_x$ le nombre de visites de la chaîne de Markov $(X_n)_{n \in \N}$ dans $x$. 
    Plus formellement : 
        $$ \boxed{V_x := \left| \left\{ n \geqslant 1 , X_n = x \right\}\right|} $$ 
    $V_x$ est donc une variable aléatoire à valeurs dans $ \overline{ \N}$. 
\end{definition}

\begin{theorem}[Nombre de visites et temps de retour]
    Soit $ \tau_x^+$ le temps de retour de la chaîne de Markov en $x \in E$. On a alors : 
    \begin{itemize} 
        \item si $ \myP( \tau_x^+ < \infty \; | \; X_0 = x) = 1$ alors $V_x = \infty$ presque sûrement. 
        \item si $ \myP( \tau_x^+ < \infty \; | \; X_0 = x) = 1 - p < 1$ alors le nombre de visites est fini 
        presque sûrement et il suit une loi géométrique. 
            \[ \text{i.e} \quad \myP(V_x = k) = p (1 - p)^k \quad \forall k \geqslant 0  \] 
    \end{itemize}
\end{theorem}

\begin{proof}
    Pour démontrer ce théorème, il serait suffisant de montrer que $ \forall k \in \N$, on ait : 
        \[ \myP(V_x \geqslant k \mid X_0 = x) = \myP( \tau_x^+ < \infty \mid X_0 = x)^k \] 
        \[ \iff \myP_x(V_x \geqslant k+1) = \myP_x( \tau_x^+) \myP_x(V_x \geqslant k) \]      
    Montrons donc que : 
        \[ \forall k \geqslant0, \quad \myP_x( V_x \geqslant k+1) = \myP_x( \tau_x^+ < \infty) \myP_x(V_x \geqslant k)  \] 
    Pour l'évènement $ \{V_x \geqslant k+1\}$ on a $V_x \geqslant 1$ donc $ \tau_x^+ < \infty$ d'où : 
        \begin{align*}
            \myP_x(V_x \geqslant k+1) &= \myP_x(V_x \geqslant k+1, \tau_x^+ < \infty) \\ 
            &= \sum_{m \geqslant 1} \myP_x(V_x \geqslant k+1, \tau_x^+ = m) \\ 
            &= \sum_{m \geqslant 1} \myP_x(V_x \geqslant k+1 \; | \; \tau_x^+ = m) \myP_x(\tau_x^+ = m)
        \end{align*}
    Or on a : 
        \begin{align*}
            & \myP_x(V_x(X_n) \geqslant k+1 \; | \; \tau_x^+ = m) \\
            &=  \myP_x(V_x(X_{n+m}) +1\geqslant k+1 \; | \; \tau_x^+ = m) \\
            &= \myP_x(V_x(X_n) \geqslant k)
        \end{align*}
    Donc : 
        \begin{align*}
            \myP_x(V_x \geqslant k+1) &= \sum_{m \geqslant 1} \myP_x(V_x \geqslant k) \myP_x( \tau_x^+ = m) \\ 
            &= \myP_x(V_x \geqslant k) \sum_{m \geqslant 1} \myP_x( \tau_x^+ = m) \\ 
            &= \myP_x(V_x \geqslant k) \times \myP_x( \tau_x^+ < \infty) 
        \end{align*}
    D'où : 
        \[ \myP_x( V_x \geqslant k) = \myP_x( \tau_x^+ < \infty)^k  \]
    Donc si $ \myP( \tau_x^+ < \infty) = 1$ alors $ \myP(V_x \geqslant k) = 1 $ pour tout $k \geqslant 0$ donc 
    $V_x = \infty$. D'autre part, si $\myP( \tau_x^+ < \infty) = 1 - p < 1$ alors $\myP(V_x \geqslant k) = (1-p)^k$. 
\end{proof}

\begin{remark}
    Précédement, nous avions vu que dans le cas où $E$ est fini et $(X_n)_{n \in \N}$ irréductible, alors : 
        \[ \mathbb{E}( \tau_x^+ \; | \; X_0 = x)  < \infty \]
        \[ \Longrightarrow \myP( \tau_x^+ < \infty \; | \; X_0 = x) = 1  \]  
    Donc on visitera le point $x$ infiniment de fois. 
\end{remark}

Nous allons maintenant chercher à découper une trajectoire en "excursions" entre deux passages successifs par le point $x$. 
D'après la propriété de Markov forte, ces excursions auront la même loi et seront indépendantes conditionnellement. 

\begin{proposition}
    Conditionnellement à l'évènement $\{ \tau_x^+ = m\}$, $(X_{n +m})_{n \in \N}$ est une chaîne de Markov 
    de condition initiale $x$ et indépendante de $X_0, \dots, X_m$. 
\end{proposition}

\subsection{État transiant et état récurrent}

\begin{definition}[État récurrent/transiant]
    Si $x \in E$ tel que $ \myP_x( \tau_x^+ < \infty) = 1$, on dit alors que $x$ est \emph{récurrent} et 
    $V_x = \infty$. Sinon, on dit que $x$ est \emph{transiant} et $ V_x < \infty$ presque sûrement. 
\end{definition}

\begin{proposition}
    On dit qu'un état $x \in E$ est \emph{récurrent} si : 
        \[ \sum_{n \geqslant 0} P^n(x,x) = \infty \] 
    Or $ P^n(x,x) = \myP(X_n = x \; | \; X_0 = x) $ et $ V_x = \sum_{n \geqslant 1} \mathbbm{1}_{X_n = x}$ donc : 
        \[ \mathbb{E}(V_x) = \sum_{n \geqslant 1} \mathbb{E}( \mathbbm{1}_{X_n = x} \; | \; X_0 = x ) = \sum_{n \geqslant 1} P^n(x,x)  \]   
    Donc si $x$ est \emph{récurrent} alors $V_x = \infty$ presque sûrement. D'où : 
        $$ \mathbb{E}(V_x)  = \infty \Longrightarrow \sum_{n \geqslant 1} P^n(x,x) = \infty $$ 
    D'autre part, si $x$ est transiant, alors $ V_x \sim \mathcal{G}(p)$ où : 
        \[ \mathbb{E}(V_x) < \infty \Longrightarrow  \sum_{n \geqslant 1} P^n(x,x) < \infty \]  
\end{proposition}

\begin{theorem}[Récurence et Transitivité]
    Soient $x,y \in E$. Alors, si $x$ est récurrent et que $ x \leadsto y$ alors $y$ est récurrent. 
\end{theorem}

\section{Mesures invariantes dans le cas dénombrable}

\subsection{États récurrents positif ou nul}

De puis le début de ce chapitre, nous nous plaçons dans un ensemble d'états $E$ fini. Or ce n'est pas toujours 
le cas. En général, $E$ est infini et dénombrable. Nous pouvions donc définir une mesure de 
probabilité invariante $ pi$ telle que : 
    \[ \forall x \in E, \quad \pi(x) = \frac{1}{ \mathbb{E}( \tau_x^+ \mid X_0 = x) }  \] 
Du fait de la dénombrabilité de E, nous aurons plusieurs classes d'états qui communiquent entre eux. 
On peut donc redéfinir la notion d'\emph{irréductibilité} pour ces chaînes. 

\begin{definition}[Chaîne irréductible (cas dénombrable)]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov homogène sur un espace d'état $E$ \emph{dénombrable}. 
    On dira que $(X_n)_{n \in \N}$ est irréductible : 
    \begin{itemize}
        \item si tous les états de $E$ sont transiants 
        \item si il existe une unique classe de récurrence dans $E$
            \[ \text{i.e} \quad \forall x,y \in E, \quad x \text{ et récurrent et } x \leadsto y \] 
    \end{itemize}
\end{definition}

\begin{remark}
    Si on considère une chaîne de Markov irréductible et récurrente alors : 
        $$ \forall x \in E, \myP_x( \tau_x^+ < \infty) = 1$$ 
    Or quand $E$ est fini, on n'a pas forcément : 
        $$  \mathbb{E}( \tau_x^+ \mid X_0 = x) < \infty $$ 
    Cela va donc nous mener à introduire une classification plus fine des états en distinguant les états récurrents 
    selon que $ \mathbb{E}( \tau_x^+ \mid X_0 = x) < \infty$ ou pas. 
\end{remark}

\begin{definition}[État récurrent positif]
    On dit qu'un état $x \in E$ est \emph{récurrent positif} s'il est récurrent et que $ \mathbb{E}( \tau_x^+ \mid X_0 = x) < \infty$. 
    On dira en revanche que $x$ est \emph{récurrent nul} s'il est récurrent et que : 
        \[ \myP_x( \tau_x^+ < \infty) = 1 \text{ et } \mathbb{E}_x( \tau_x^+) = \infty \] 
\end{definition}

La notion de récurrence positive ou nulle permet de quantifier la vitesse de retour de la chaîne dans un état $x$. 
Un état récurrent positif est donc un état où la chaîne revient souvent alors qu'un état récurrent nul est un état où 
la chaîne revient en temps fini mais potentiellement très long. 
C'est ce que traduit le fait que $ \mathbb{E}_x( \tau_x^+)$ soit fini ou pas. 

\begin{remark}
    Dans le cas des espaces d'états finis, lors de la construction de la mesure de probabilités invariante, 
    nous avions posé : 
        \[ \overline{ \pi}(y) = \mathbb{E} \left( \sum_{n=0}^{ \tau_x^+ -1} \mathbbm{1}_{X_n} = y \right) \] 
    en tant que mesure invariante de masse totale $ \mathbb{E}_x( \tau_x^+)$. Or dans le cas actuel si $(X_n)_{n \in \N}$ 
    est réucurrente positive, on aura $ \mathbb{E}_x( \tau_x^+) < \infty$ donc on pourra construire notre 
    mesure de probabilités invariante. Dans le cas contraire, notre mesure ne sera pas de probabilité.  
\end{remark}

\subsection{Propriétés des chaînes irréductibles}

\begin{prop}[Positivité]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov irréductible récurrente.  
    Si $\pi$ est une mesure de probabilités invariante non nulle, alors : 
        \[
            \forall x \in E, \quad \pi(x) > 0.
        \]
\end{prop}

\begin{proof}
    Comme $\pi$ est non nulle, il existe $x_0 \in E$ tel que $\pi(x_0) > 0$.  
    Comme la chaîne est irréductible, pour tout $y \in E$, il existe $n_0 \ge 0$ tel que :
        \[ P^{n_0}(x_0, y) > 0 \]
    En utilisant l’invariance de $\pi$ et en isolant le terme $z = x_0$, on obtient :
        \[ \pi(y) = \sum_{z \in E} \pi(z) P^{n_0}(z, y) \geqslant \pi(x_0) P^{n_0}(x_0, y) > 0 \]
    Ainsi, chaque état $y \in E$ a une mesure strictement positive sous $\pi$.
\end{proof}

\begin{theorem}[Unicité]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov irréductible et récurrente. Il existe alors une unique 
    mesure invariante à constante multiplicative près. 
    \begin{itemize}
        \item De plus, si tous les états sont \emph{récurrents nuls} alors toute mesure invariante non nulle est de masse infinie. 
        \item En revanche, si $(X_n)_{n \in \N}$ est \emph{récurrente positive}, alors toutes les mesures invariantes sont 
        de masse finie donc il existe une unique mesure de probabilités invariante telle que : 
            \[ \forall x \in E, \quad \pi(x) = \frac{1}{ \mathbb{E}_x( \tau_x^+) }  \] 
    \end{itemize}
\end{theorem}

\begin{example}[Modèle de file d'attente]
    Soit $(X_n)_(n \in \N)$ une chaîne de Markov définie par le graphe suivant :

                    (2) edge[above, bend right] node{$1-p$} (1)

            \begin{cases}
                P(k,k+1) = p \quad \text{si } k \geqslant 1 \\ 
                P(k, k-1) = 1 - p \quad \text{si } k \geqslant 1 \\

    Cette chaîne de Markov est irréductible car : 
        \[ \forall x,y \in \N, \exists n \in \N, \quad \myP_x(X_n = y) > 0  \] 
        \[ \iff \forall x,y \in \N, \exists n \in \N, \quad P^n(x,y) > 0 \]
    Déterminons une mesure invariante pour cette chaîne de Markov. Soit $ \pi$ une mesure invariante pour $P$. 
    Alors elle vérifie $ \pi P = \pi$ d'où : 
        \[ 
            \begin{cases}
                \displaystyle
                \pi(0) = \sum_{k \in \N} \pi(k) P(k,0) = (1-p) \pi(1) \\[1em]
                \displaystyle
                \pi(1) = \pi(0) + (1-p) \pi(2)
            \end{cases}

    et $ \forall k \geqslant 1$, on a $ \pi(k) = p \pi(k-1) + (1-p) \pi(k+1)$. Nous avons donc une relation 
    de récurrence d'ordre 2 : 
        $$ Q = (1-p)X^2 - X + p $$ 
    de racines évidentes $1$ et $p/(1-p)$. Les solutions de la relation de récurrence sont donc de la forme : 
            \[ 
                \forall k \geqslant 1, \quad \pi(k) = a 1^k + b \left( \frac{p}{1-p} \right)^k 
            \] 
    Déterminons $a$ et $b$. On a : 
            \[ 
                \begin{cases}
                    \pi(1) = \dfrac{ \pi (0)}{1-p} = a + \dfrac{bp}{1 - p} \\[1em]
                    \pi(2) = \dfrac{ \pi(0)}{1-p} \left( \dfrac{1}{1-p} -1 \right) = \dfrac{ \pi(0) p}{(1-p)^2} = a + b \dfrac{p^2}{(1-p)^2}      
                \end{cases}
            \] 
    Par identification, on obtient donc : 
        \[ a = 0  \quad \text{et} \quad b = \frac{ \pi(0)}{p}  \]
    Donc la solution de l'équation de récurrence s'écrit : 
        \[  \forall k \geqslant 1, \quad \pi(k) = \frac{ \pi(0)}{p} \left( \frac{p}{1-p} \right)^k  \] 
    Pour aller plus loin, en considérant la série $ \sum_{k \geqslant 0} \pi(k)$ on peut affirmer que 
    si $p < 1/2$ alors la mesure peut être de probabilité. Dans le cas contraire, toutes les mesures invariantes seront 
    de masse infinie. 
\end{example}

\subsection{Critère de récurrence positive}

Il peut être très difficile de calcule $ \mathbb{E}_x( \tau_x^+)$ car il faut très bien connaître 
les propriétés des excursions de la chaîne. En pratique, on utilisera le critère suivant : 

\begin{theorem}[Réciproque]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov irréductible. Si il existe une mesure invariante $ \pi$ non nulle 
    et de masse finie alors la chaîne est récurrente positive. 
\end{theorem}

Ce théorème traduit le fait qu'il y a équivalence entre le fait qu'une chaîne de Markov soit récurrente positive et 
qu'elle admette une mesure de probabilités invariante. 

\begin{proof}
    Revenons à notre théorème d'unicité des mesures invariantes. Le principe de la démonstration est le même que 
    dans le cas discret. Soit $ \pi$ une mesure invariante non nulle de masse finie. On veut comparer $ \pi$ 
    avec $ \overline{ \pi}$ qui compte le nombre moyen de passage en $y \in \E$. 
        \[ \forall x \in E, \quad \overline{ \pi}(x) = \mathbb{E}_x \left( \sum_{n=0}^{ \tau_x^+ - 1} \mathbbm{1}_{X_n = y}  \right) = \sum_{n \geqslant 0} \myP_x( X_n = y, \tau_x^+ > n) \] 
    or $ \pi$ est invariante donc $ \forall y \in E$, on a : 
        \begin{align*}
            \pi(y) &= \sum_{z \in E} \pi(z) P(z,y) \\
            &= \pi(x)P(x,y) + \sum_{z_1 \not = x} \pi(z_1)P(z_1, y) \\ 
            &= \pi(x) P(x,y) + \sum_{z_1 \not = x} \left( \sum_{z_2 \in E} \pi(z_2)P(z_2, z_2) \right) P(z_1, y) \\ 
            &= \pi(x) P(x,y) + \pi(x) \underbrace{\sum_{z_1 \not = x} P(x,z_1)P(z_1,y)}_{ \text{terme } z_2 = x } 
            + \sum_{z_1 \not = x} \left( \sum_{z_2 \not = x} \pi(z_2) P(z_2, z_1) P(z_1, y)\right) 
        \end{align*}
    où $ \pi(x) \sum_{z_1 \not = x} P(x,z_1)P(z_1,x) = \myP(X_2 = y, X_1 \not = x, X_0 = x)$. Si on itère $ l \in \N$ 
    fois ce processus, on aura donc : 
        \begin{align*}
            \pi(y) &= \pi(x) \left( P(x,y) + \sum_{k=1}^{l-1} \sum_{\substack{z_1 \not = x \\ \dots \\ z_k \not = x}} P(x,z_k) P(z_k, z_{k-1}) \dots P(z_1, y)\right) \\ 
            & + \sum_{\substack{z_1 \not = x \\ \dots \\ z_l \not = x}} \pi(z_l) P(z_l, z_{l-1}) \dots P(z_1, y) 
        \end{align*}
    Donc 
        \[ \pi(y) \geqslant \pi(x) \left[ \sum_{k=1}^{l-1} \myP_x(X-k = y, \tau_x^+ > k ) \right] \]
    or ceci est valable pour tout $l \in \N$ donc on peut faire tendre $l$ vers $\infty$ d'où : 
        \[ \pi(y) \geqslant \pi(x) \overline{ \pi}(y) \] 
    Si on choisit $x \in E$ tel que $ \pi(x) > 0$ alors : 
        \[ \infty > \frac{ \pi(y)}{ \pi(x)} \geqslant \overline{ \pi}(y) \] 
    donc $ \overline{ \pi}$ est bien définie. Or $ \overline{ \pi}$ est définie de telle sorte que : 
        \[ \sum_{y \in E} \overline{ \pi}(y) = \mathbb{E}_x( \tau_x^+) \leqslant \sum_{y \in E} \frac{ \pi(y)}{ \pi(x)}   \] 
    or $ \sum_{y \in E} \frac{ \pi(y)}{ \pi(x)} < \infty$ car $ \pi$ est finie donc : 
        \[ \forall x \in E, \quad \mathbb{E}_x( \tau_x^+) < \infty  \] 
    i.e $x$ est récurrent positif donc la chaîne des récurrente potivite irréductible. On peut donc construire l'unique 
    mesure de probabilité invariante. 
\end{proof}
