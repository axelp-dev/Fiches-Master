% Introduction 

L'algorithme de Metropolis-Hasting est une procédure permettant de transformer une chaîne de Markov par le rejet 
sélectif de certains de ses pas en une nouvelle chaîne admettant une mesure invariante fixée. 

\begin{proposition}[Algorithme de Metropolis-Hastings]
    Soit $(U_n)_{n \ge 1}$ et $(V_n)_{n \ge 0}$ deux suites de variables aléatoires indépendantes et identiquement distribuées,  
    telles que $U_n \sim \mathcal{U}([0,1])$ et $V_n \sim \mathcal{U}([0,1])$.  

    Soit $f : E \times [0,1] \longrightarrow E$ une fonction qui génère un candidat $y = f(x,U)$ à partir d'un état $x \in E$ et d'une variable uniforme $U$.  

    On définit la chaîne $(Y_n)_{n \geqslant 0}$ de Metropolis-Hastings par :
        \[
            Y_0 \in E, \quad
            Y_{n+1} =
            \begin{cases}
                y = f(Y_n, U_{n+1}), & \text{si } V_n < \alpha(Y_n, y), \\
                Y_n, & \text{sinon},
            \end{cases}
        \]
    où la probabilité d'acceptation est
        \[
            \alpha(x,y) = \min\left(1, \frac{\pi(y) Q(y,x)}{\pi(x) Q(x,y)}\right)
        \]
    et
        \[
            Q(x,A) = \mathbb{P}(f(x,U) \in A), \quad A \subset E.
        \]
\end{proposition}

\section{Cas d'un espace d'états fini}

\begin{proposition}[Matrice de transition]
    Soit $(X_n)_{n \in \N}$ une chaîne de Markov sur un espace d'états fini $E$. On note : 
        $$ \forall x,y \in E, \quad Q(x,y) = \myP(X_1 = y \mid X_0 = x) $$
    Supposons que $(X_n)_{n \in \N}$ est irréductible. Alors l'algorithme de Metropolis-Hasting 
    transforme la chaîne $(X_n)_{n \in \N}$ en une chaîne $(Y_n)_{n \in \N}$ de matrice de transition : 
        $$ P(x,y) = 
            \begin{cases}
                Q(x,y) \times \min \left( 1, \frac{ \pi(y) Q(y,x)}{ \pi(x) Q(x,y)} \right) \quad \text{si} \quad n \not = y \\ 
                1 - \sum_{z \not = x} P(x,z) \quad \text{si} \quad x = y  
            \end{cases}
        $$ 
\end{proposition}

\begin{proof}
    Soient $x,y \in E$ tels que $x \not = y$. Calculons $ \myP(Y_1 = y \mid Y_0 = x)$. On a : 
        \begin{align*}
            \myP( Y_1 = y \mid Y_0 = x) &= \myP_x \left( f(x,U_1) = y, V_1 < \frac{ \pi(y) Q(y,x)}{ \pi(x) Q(x,y)} \right)  \\
            &= \myP_x(X_1 = y) \myP \left( V_1 < \frac{ \pi(y) Q(y,x)}{ \pi(x) Q(x,y)}\right) \\
            &= Q(x,y) \min \left( 1, \frac{ \pi(y) Q(y,x)}{ \pi(x) Q(x,y)} \right)
        \end{align*} 
    De même : 
        \begin{align*}
            \myP(Y_1 = \mid Y_0 = x) &= 1 - \sum_{y \not = x} \myP(Y_1 = y \mid Y_0 = x) \\
            &= 1 - \sum_{y \not = x} P(x,y)
        \end{align*}
\end{proof}

\begin{proposition}
    Si pour tout $x,y \in E$, on a : 
        $$ Q(x,y) > 0 \iff Q(y,x) > 0 $$ 
    alors la mesure $ \pi$ est réversible pour la chaîne $(Y_n)_{n \in \N}$. 
\end{proposition}

\begin{proof}
    Soient $X,y \in E$. On souhaite vérifier que : 
        $$ \pi(x)P(x,y) = \pi(y)P(y,x) $$ 
    Distinguons deux cas : 
    \begin{itemize}
        \item \underline{si $x = y$} alors on a évidement $\pi(x)P(x,x) = \pi(x)P(x,x)$. 
        \item \underline{si $x \not = y$} alors : 
            $$ \pi(x) P(x,y) = \pi(x) Q(x,y) \min \left(1, \frac{ \pi(y) Q(y,x)}{ \pi(x) Q(x,y)}  \right) $$ 
        de même : 
            $$ \pi(y) P(y,x) = \pi(y) Q(y,x) \min \left(1, \frac{ \pi(x) Q(x,y)}{ \pi(y) Q(y,x)}  \right) $$ 
        Supposons que $  \pi(x) Q(x,y) \leqslant \pi(y) Q(y,x)$ alors : 
            $$ \pi(x)P(x,y) = \pi(x) Q(x,y) $$ 
        et 
            \begin{align*}
                \pi(y) P(y,x) &= \pi(y) Q(y,x) \frac{ \pi(x) Q(x,y)}{ \pi(y) Q(y,x)} \\ 
                &= \pi(x) P(x,y) 
            \end{align*}
        Si $ \pi(x) Q(x,y) > \pi(y) Q(y,x) $, le raisonnement est le même.
    \end{itemize}
\end{proof}

\begin{example}
    Soit $(X_n)_{n \in \N}$ la marche aléatoire simple sur $\Z$. Soit $ \pi$ une mesure de probabilité sur $ \Z$. 
    On cherche à construire une chaîne de Markov $(Y_n)_{n \in \N}$ sur $\Z$ avec l'algorithme de Metropolis-Hasting 
    de mesure réversible $ \pi$. On a donc les algorithmes suivants : 

    \begin{center}
        \begin{minipage}{0.45\textwidth}
            \begin{lstlisting}[language=Python]
def etapeMa(x): 
    if rand() < 1/2: 
        return x+1
    return x-1
            \end{lstlisting}
        \end{minipage}
        \hfill 
        \begin{minipage}{0.45\textwidth}
            \begin{lstlisting}[language=Python]
def algoMetHast(x): 
    y = etapeMa(x)
    if rand() < pi(y) * (1/2) / pi(x) * (1/2): 
        return y 
    return x
            \end{lstlisting}
        \end{minipage}
    \end{center}

    On a donc : 
        $$ Q(x,x+1) = Q(x,x-1) = \frac{1}{2} $$ 
        $$ \forall x,y \in E, \; |x-y| \not = 1, \quad Q(x,y) = 0 $$ 
    Et : 
        $$ P(x,x+1) = \frac{1}{2} \min \left( 1, \frac{ \pi(x+1)}{ \pi(x)}  \right) \quad \text{et} \quad P(x, x-1) = \frac{1}{2} \min \left(1, \frac{ \pi(x-1)}{ \pi(x)} \right) $$
\end{example}

\begin{prop}[Irréductibilité et apériodicité]
    Soit $Q$ une matrice de transition irréductible sur un espace d'états $E$ tel que
        $$ \forall x, y \in E, \quad Q(x,y) > 0 \iff Q(y,x) > 0, $$
    et soit $\pi$ une distribution cible strictement positive sur $E$.  
    Soit $P$ la matrice de transition de la chaîne de Markov construite par l'algorithme de Metropolis-Hastings à partir de $Q$ et $\pi$.  

    Alors :
        \begin{enumerate}[label=\roman*)]
            \item La chaîne de transition $P$ est irréductible.
            \item Si $P(x,x) > 0$ pour au moins un $x \in E$, alors $P$ est apériodique.
        \end{enumerate}
\end{prop}

\begin{proof}
    \begin{enumerate}[label=\roman*)]
        \item \textbf{Irréductibilité.}  
            Soient $x, y \in E$. Comme $Q$ est irréductible, il existe un chemin fini
            $$ x = z_1, z_2, \dots, z_k = y \in E $$
            tel que $Q(z_i, z_{i+1}) > 0$ pour tout $i = 1, \dots, k-1$.  
            Or par définition de Metropolis-Hastings :
            $$ P(z_i, z_{i+1}) = Q(z_i, z_{i+1}) \min \left( 1, \frac{\pi(z_{i+1}) Q(z_{i+1}, z_i)}{\pi(z_i) Q(z_i, z_{i+1})} \right) > 0, $$
            donc $z_1, \dots, z_k$ constitue un chemin admissible pour $P$. Ainsi, $P$ est irréductible.
            
        \item \textbf{Apériodicité.}  
            Pour tout état $x \in E$, la probabilité de rester en $x$ est
            $$ P(x,x) = 1 - \sum_{y \neq x} P(x,y) \ge 0. $$
            Si $P(x,x) > 0$ pour au moins un $x$, alors la chaîne peut rester en $x$ avec une probabilité strictement positive, ce qui assure l'apériodicité de $P$.
    \end{enumerate}
\end{proof}

\begin{remark}[Cas des chaînes symétriques]
    Si $Q$ est une matrice symétrique. 
        $$ \text{i.e} \quad \forall x, y \in E, \quad Q(x,y) = Q(y,x) $$ 
    et si elle est irréductible, alors l'algorithme de Metropolis-Hasting s'exprime de façon 
    simplifiée : 
        $$ P(x,y) = Q(x,y) \min \left(1, \frac{ \pi(y)}{ \pi(x)}\right) \quad \forall x,y \in E $$ 
    et : 
        $$ P(x,y) = 1 -  \sum_{z \neq x} P(x,z) \quad \text{si } x = y $$   
    Dans ce cas, la probabilité d'acceptation de la transition $ x \to y$ est $ 1 - \min \left(1, \pi(y) / \pi(x)\right)$. 
\end{remark}

\section{Cas d'un espace d'états continu}

\begin{proposition}
    Dans le cas où $E$ est un espace continu, tout fonctionne de manière analogue.  
    On note $Q(x,y)$ la densité de la variable $X_1$ sachant $X_0 = x$. Alors :
        \[
            \mathbb{E}(\varphi(X_1) \mid X_0 = x) = \int_E \varphi(y) Q(x,y) \, dy, \quad \forall \varphi \text{ mesurable}.
        \]

    Pour tout petit intervalle $[y, y+\Delta_y]$, on a :
        \[
            \myP(X_1 \in [y, y+\Delta_y] \mid X_0 = x) \simeq Q(x,y) \Delta_y.
        \]

    On note $\pi(x)$ la densité de la mesure cible et on définit la transition de la chaîne $(Y_n)_{n \in \N}$ par :
        \[
            P(x,dy) = Q(x,y) \min\left(1, \frac{\pi(y) Q(y,x)}{\pi(x) Q(x,y)}\right) dy
            + \left[1 - \int_E Q(x,y) \min\left(1, \frac{\pi(y) Q(y,x)}{\pi(x) Q(x,y)}\right) dy \right] \delta_x(dy),
        \]
    où $\delta_x$ est la mesure de Dirac en $x$.  
    On a alors :
        \[
            \int_E P(x,dy) = 1.
        \]
    L'algorithme reste lui aussi inchangé. On génère $y = X_1$ à, partir de la chaîne $Q$. On accepte la transition 
    avec probabilité $\frac{\pi(y) Q(y,x)}{\pi(x) Q(x,y)}$. Si on refuse la transition, on reste sur place. 
    À nouveau, la chaîne de Markov sera, sous de bonnes hypothèses, irréductible et la mesure $ \pi$ sera réversible 
    pour cette chaîne. 
\end{proposition}

\begin{example}[Marche aléatoire à pas gaussiens]
    Soit $(Z_n)_{n \in \N}$ une suite de variables iid de loi $ \mathcal{N}(0,1)$. On définit 
    le processus $(X_n)_{n \in \N}$ par : 
        $$ X_0 \in \R, \quad \forall n \geqslant 1, X_{n+1} = X_n + Z_{n+1} $$ 
    On aura donc : 
        $$ \forall x,y \in \R, \quad Q(x,y) = \frac{1}{\sqrt{2 \pi}} \exp \left( - \frac{(y-x)^2}{2}\right) $$ 
    Donc si $ \pi$ est une densité cible quelconque, alors : 
        $$ \frac{\pi(y) Q(y,x)}{\pi(x) Q(x,y)} = \frac{ \pi(y)}{ \pi(x)} \quad \text{car} \quad Q(x,y) = Q(x,y) $$ 
    On peut alors définir la nouvelle chaîne $(Y_n)_{n \in \N}$ par : 
        $$ \forall n \in \N, Y_{n+1} = Y_n + Z_{n+1} \mathbbm{1}_{ \frac{ \pi(Y_n + Z_{n+1})}{ \pi(Y_n)} < V_{n+1}} $$ 
    avec $(V_n)_{n \in \N}$ une suite idd de variables de loi $ \mathcal{U}([0,1])$. Cette nouvelle 
    chaîne de Markov admet alors $ \pi$ pour mesure invariante, réversible et ergodique. 
    De plus : 
        $$ Y_n \overset{\text{L}}{ \underset{n \to \infty}{\longrightarrow} } \pi $$ 
\end{example}

L'algorithme de Metropolis-Hasting permet donc d'approcher n'importe quelle densité sur $\R$. 
Or il peut être très lent dans certains cas.

\section{Mesures de Gibbs sur un espace d'états fini}

Soit $E$ un espace d'états fini. On définit : 
    \begin{equation}
        H : E \longrightarrow \R
    \end{equation}
une fonction d'énergie sur $E$. En physique statistique, elle représenterait 
l'énergie d'un état du système. 

\begin{definition}[Mesure de Gibbs]
    La mesure de Gibbs sur $E$ de température inverse $ \beta$ est la mesure 
    de probabilité $ \mu_{ \beta}$ définie par : 
        \begin{equation}
            \mu_\beta(x) = \frac{1}{z_\beta} e^{ \beta H(x) } 
        \end{equation}
    où $z_\beta = \sum_{x \in E} e^{ \beta H(x)}$. 
\end{definition}

\begin{proposition}[Fondamentale]
    Par construction de la mesure de Gibbs, lorsque $\beta$ est petit (le système a beaucoup d'énergie),
    la mesure a tendence à s'uniformiser sur $E$. Or, lorsque $ \beta$ est petit (i.e le système a peu d'énergie), 
    alors $ \mu_\beta$ ne charge plus que l'ensemble $ \mathcal{M} = \{x \in E \mid \mu_\beta(x) = \mu_0\}$ où 
    $ \mu_0$ est le maximum de $ \mu_\beta$. 

    \medskip 

    L'algorithme du recuit simulé, que nous verrons plus tard, cherche donc à déterminer de telles solutions en 
    simulant une marche aléatoire sur $E$ de loi $ \mu_\beta$. En reffroidissant le système petit à petit (i.e en augmentant 
    $ \beta$), la marche se stabilisera sur un maximum. Mais pour cela, nous devons savoir simuler $ \mu_\beta$. 
    Pour cela, on utilisera l'algorithme de Metropolis-Hastings.
\end{proposition}

Or lorsque $E$ est grand, simuler une variable aléatoire de loi $ \mu_\beta$ 
est difficile pour plusieurs raisons : 
\begin{itemize}
    \item $z_\beta$ devient très grand : lors de l'approximation numérique, 
    on peut donc avoir des valeurs de $ e^{ \beta H(x)}/z_\beta$ encodées
    comme $0$. 
    \item $z_\beta$ devient difficile à calculer du fait de la taille de $E$.
\end{itemize}

Les méthodes directes de simulation de mesure de Gibbs fonctionnent très mal. 
Heureusement, on utilise alors l'algorithme de Metropolis-Hasting.

\begin{remark}[Adaptation de l'algorithme]
    Soit $Q$ la matrice de transition d'une chaîne de Markov irréductible 
    sur $E$ telle que $Q(x,y) > 0 \iff Q(y,x) > 0$ pour tout $x,y \in E$.
    Pour simuler la loi $ \mu_\beta$, on utilise la matrice de transition 
    définie pour tout $x,y \in E$ par : 
        \begin{align}
            P_\beta (x,y) &= Q(x,y) \min \left( 1, \frac{ \mu_\beta (y) Q(y,x)}{ \mu_\beta(x) Q(x,y)}\right) \\ 
            &= Q(x,y) \min \left( \frac{ \frac{e^{ \beta H(y)}}{ z_\beta} Q(y,x)}{ \frac{e^{ \beta H(x)}}{ z_\beta} Q(x,y)} \right) \\ 
            &= Q(x,y) \min \left( 1, e^{\beta (H(y) - H(x))} \frac{Q(y,x)}{Q(x,y)}\right) \label{eq:calcul-annulation-gibbs}
        \end{align}
    On pourra donc appliquer l'algorithme sans problème d'approximation 
    de très petites valeurs, car $z_\beta$ n'apparaît plus.
\end{remark}

\begin{example}[Depinning Transition]
    Soit l'espace d'états suivant : 
        $$ E = \left\{ \text{chemins au plus proche voisin sur } \Z \text{ de longueur } n \right\} $$ 
    Un élément de $E$ est donc une marche aléatoire sur $E$. On définit la fonctionnelle 
    d'énergie $H$ sur $E$ telle que : 
        $$ \forall s \in E, \quad H(s) = \text{Card}(\{0 \leqslant i \leqslant n \mid s_i = 0\}) $$ 
    On souhaite simuler un élément de $E$ sous la loi $ \mu_\beta$. On peut proposer
    différentes transformations pour un chemin de $E$ : 
    \begin{enumerate} 
        \item Si $s \in E$, on tire $i \in \llbracket 1, n \rrbracket$ on remplace 
        ensuite le saut en $i$ par le saut opposé. 
        \emph{(C'est une marche réversible).}
        \item Si $s \in E$, on tire $i \in \llbracket 1, n \rrbracket$ et 
        on remplace un schéma $ \bigvee $ en $i$ par un $ \bigwedge $ en $i$.
         \emph{(Celle-ci n'est pas réversible.).}
    \end{enumerate}
    D'après \ref{eq:calcul-annulation-gibbs}, on peut donc définir l'algorithme suivant : 

    \begin{lstlisting}[language=Python]
def etapeP(x,b):
    y = etapeQ(x)
    if rand() < (Q(y,x) / Q(x,y)) * exp(b (H(y) - H(x))): 
        return y
    return x
    \end{lstlisting}
    
    Le comportement de la chaîne ainsi produite sera donc déterminé par 
    le paramètre $ \beta$. 
    Si $ \beta$ sera faible (i.e température basse), la chaîne aura un comportement 
    de marche aléatoire symétrique. En revanche, autour d'une certaine valeur 
    de $ \beta$ appelée $ \beta_c$, dit critique, la chaîne orbitera autour 
    de l'axe des abscisses. C'est ce que l'on appelle une \emph{transition de phase}.
\end{example}

\begin{remark}
    Si $Q$ est une matrice \emph{réversible et symétrique} alors 
        $$ \forall x,y \in E, P_\beta(x,y) = Q(x,y) \min \left(1, e^{\beta (H(y) - H(y))}\right)$$

    Dans ce cas, plutôt que de tester \texttt{rand() < (Q(y,x) / Q(x,y)) * exp(b (H(y) - H(x)))}, 
    on préfère tester \texttt{log(rand()) < b (H(y) - H(x))}. Qui sera moins 
    sujet à provoquer des erreurs d'approximation (de valeurs infinies).
\end{remark}

\section{Algorithme du recuit simulé}

On se place toujours dans un espace d'états $E$ fini et $H : E \longrightarrow \R$ 
une fonction.
On souhaite déterminer : 
    $$ 
        \begin{cases}
            \max_E H \\ 
            \text{argmax}_E H 
        \end{cases}
    $$
La méthode du recuit simulé permet de déterminer un estimateur de ces quantités 
pour un coût relativement faible par rapport à une exploration complète de l'espace 
d'états. Elle est basée sur la remarque suivante.

\begin{lemma}
    Soit $E^*$ l'ensemble définit par : 
        \begin{equation}
            E^* := \{ e \in E \mid H(e) = \max_E H\}
        \end{equation}
    On a $\lim_{\beta \to \infty} \mu_\beta \sim \mathcal{U}(E^*)$.
\end{lemma}

\begin{proof}
    Soit $ \delta > 0$ tel que $H(e) < \max H - \delta$ pour tout $e \not \in E^*$.
    On a donc : 
        $$ \sum_{e \in E^*} e^{ \beta H(e)} \leqslant z_\beta = \sum_{e \in E} e^{ \beta H(e)} 
        = \sum_{e \in E^*} e^{ \beta H(e)} + \sum_{e \in E/E^*} e^{ \beta H(e)} $$ 
        $$ \iff |E^*| e^{ \beta \max H} \leqslant z_\beta \leqslant |E^*| e^{\beta \max H} + |E| e^{ \beta(\max H - \delta)} $$ 
    Donc pour $ e \in E^*$, on a :
        \begin{align*}
            & \mu_\beta (e) = \frac{e^{ \beta \max H}}{z_\beta} \\ 
            \iff \frac{e^{ \beta \max H}}{e^{ \beta \max H} |E^*| + |E| e^{ \beta (\max H - \delta)}} \leqslant & \mu_\beta (e) \leqslant \frac{e^{ \beta \max H}}{e^{ \beta \max H} |E^*|} \\ 
        \end{align*}
    donc $ \frac{1}{ |E^*| + |E| e^{- \beta \delta}} \leqslant \mu_\beta (e) \leqslant \frac{1}{|E^*|}$. 
    D'où $ \lim_{ \beta \to \infty} \mu_\beta(e) = 1/|E^*|$. 

    De même pour $ e \in E/E^*$, on a : 
        $$ \mu_\beta (e) \leqslant \frac{e^{ \beta (\max H - \delta)}}{ | E^*| e^{ \beta \max H}} \leqslant \frac{e^{- \beta \delta}}{|E^*|} \underset{\beta \to \infty}{\longrightarrow} 0 $$
    d'où $ \mu_\beta (e) \underset{\beta \to \infty}{\longrightarrow} 1/|E^*| \mathbbm{1}_{e \in E^*}$. 
\end{proof} 

L'algorithme du recuit simulé consite donc à utiliser à chaque étape 
l'algorithme de Metropolis-Hasting pour simuler $ \mu_\beta$ avec 
$ \beta_n \underset{n \to \infty}{\longrightarrow} \infty$ assez lentement. 

\begin{proposition}[Algorithme du recuit simulé]
    On se donne \texttt{etape Q}, et une suite $( \beta_n)_{n \geqslant 0}$. 
    On définit ensuite : 
    \begin{lstlisting}[language=Python]

def recuitSimule(x,n): 
    y = etapeQ(x)
        if rand() < exp(beta[n] (H(y) - H(x))) * (Q(y,x) / Q(x,y)): 
            return y 
        return x
    \end{lstlisting}
    La différence fondamentale de cet algorithme est que la probabilité de 
    transition dépend maintenant de $n$.
\end{proposition}

\begin{theorem}[Convergence de l'algorithme du recuit simulé]
    On note $(z_n)_{n \geqslant 0}$ la chaîne de Markov inhomogène (la chaîne dépend de l'état 
    précédent et de la posittion dans le temps $n$) dans 
    le temps de recuit simulé. Il existe $c_* > 0$ tel que pour tout $ c < c_*$,
    en posant $ \beta_n := c \log n$, alors : 
        \begin{equation}
            \boxed{ \lim_{n \to \infty} z_n \in E^*} \quad \text{p.s} 
        \end{equation}
    \emph{
        Ici, la suite $(\beta_n)_{n \geqslant 0}$ est la suite des températures inverses que nous utilisons pour simuler 
        le reffroidissement du système lors de l'exécution de l'algorithme.
        Ce théorème nous assure donc qu'il existe une courbe de reffroidissement optimale qui permet la 
        convergence de l'algorithme.
    }
\end{theorem}


\begin{remark}
    Ce théorème nous assure que si l'on refroidit petit à petit le système (i.e $ n \to \infty$), 
    alors la chaîne tend à s'arrêter sur un maximum de $H$.
    Dans cette situation, $ \hat{M}_n = \max_{k \leqslant n} H(z_k)$ et 
    $ \hat{E}_n = \text{argmax}_{z \in \{z_k \mid k \in \N\}} H(z)$ sont des estimateurs 
    fortement consistants de $\max H$ et $ \text{argmax } H$.

    \medskip 

    En pratique ces estimateurs convergent très lentement pour un tel choix de 
    $ \beta$. On préfèrera donc faire tendre $ \beta$ vers $ \infty$ plus rapidement 
    et simuler plusieurs chaînes partant de points différents de l'espace d'état.

    \medskip 

    Cet algorithme est très performant en plusieurs dimensions puisque l'algorithme 
    ne dépend pas de la nature des données sur lequelles sont calculées $H$.
    En revanche, il est peu performant pour des fonctions $H$ trop périodiques.
\end{remark}
